<?php

require_once __DIR__."/../../vendor/autoload.php";
use Museum\Object\Blog;

// Chỉnh sửa phần xử lý xóa ảnh
if (isset($_POST['id'])) {
    $id = $_POST['id'];

    // Lấy blog từ ID
    $blog = Blog::getById($id);
    
    if ($blog && $blog->imgUrl) {
        // Kiểm tra đường dẫn file ảnh
        $imagePath = $_SERVER['DOCUMENT_ROOT'] . '/museum-website/' . $blog->imgUrl;  // Đảm bảo đường dẫn tuyệt đối

        if (file_exists($imagePath)) {
            // Xóa ảnh trên server
            if (unlink($imagePath)) {
                // Cập nhật cơ sở dữ liệu để xóa ảnh
                $blog->imgUrl = null; // Hoặc update trường ảnh thành null
                if (Blog::add($blog)) { // Kiểm tra việc lưu vào cơ sở dữ liệu
                    // Phản hồi JSON hợp lệ
                    echo json_encode(['success' => true, 'message' => 'Image deleted and database updated successfully.']);
                } else {
                    // Phản hồi JSON hợp lệ khi không thể cập nhật cơ sở dữ liệu
                    echo json_encode(['success' => false, 'message' => 'Failed to update database.']);
                }
            } else {
                // Phản hồi JSON hợp lệ khi không thể xóa ảnh từ server
                echo json_encode(['success' => false, 'message' => 'Failed to delete image from server.']);
            }
        } else {
            // Phản hồi JSON hợp lệ khi ảnh không tồn tại trên server
            echo json_encode(['success' => false, 'message' => 'Image does not exist on the server.']);
        }
    } else {
        // Phản hồi JSON hợp lệ khi không tìm thấy blog hoặc không có ảnh
        echo json_encode(['success' => false, 'message' => 'Blog not found or no image to delete.']);
    }
}

?>
